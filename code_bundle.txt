===== README.md =====

# Quant

## News API setup

This project includes a free-tier news provider adapter using Finnhub.

1. Copy the example environment file:
   - `copy .env.example .env`
2. Add your key:
   - `FINNHUB_API_KEY=your_key_here`
3. Use a user-maintained ticker universe in `RESEARCH_SYMBOLS` if you want to change the defaults.

Finnhub requests are made only when a refresh or quote fetch is explicitly run. If no key is configured, live price refresh is unavailable.

## Public news feeds

The provider registry also supports the public-feed approach used by
[`trader-news`](https://github.com/Jakeschincariol/trader-skill): Yahoo Finance
headline RSS for an explicitly supplied ticker list, plus SEC EDGAR 8-K feeds
when a valid contact User-Agent is configured. Headlines and filings are
normalized into the existing SQLite news-event pipeline; important 8-K items
are marked in their titles. These are informational sources, not trading
recommendations.

Set `SEC_USER_AGENT` to a real contact string, for example
`Quant Research your.name@example.com`, to enable SEC requests. Without it,
the public provider still fetches Yahoo RSS and skips SEC filings. The provider
does not make requests until `fetch_live_news` is called with symbols.

## Autonomous research cycle

Run one live refresh and research cycle:

```powershell
.\.venv\Scripts\python.exe -m src.autonomous --top 10
```

Keep the process running for hourly cycles:

```powershell
.\.venv\Scripts\python.exe -m src.autonomous --watch --interval 3600
```

Each cycle refreshes up to 90 days of daily candles for `RESEARCH_SYMBOLS`,
selects the most volatile symbols, fetches their news, generates research
signals, resolves matured outcomes, and reports a chronological out-of-sample
evaluation with 5 basis points of estimated cost per side. The evaluator is a
diagnostic for the fixed strategies, not a forecast or investment recommendation.
News can adjust confidence slightly but does not reverse a strategy's signal.

The simulated broker can persist cash, positions, and order IDs across
restarts when constructed with `state_db_path="data/paper_broker.db"`. This is
opt-in; without it, broker state remains in memory. It still does not submit
real orders.

class PriceRefresher(Protocol):
    def refresh(
        self,
        csv_path: str | Path,
        symbols: Iterable[str],
        *,
        start: datetime,
        end: datetime,
    ) -> int: ...


NewsFetcher = Callable[..., list[dict[str, object]]]


def run_autonomous_cycle(
    *,
    db_path: str | Path = DEFAULT_DB_PATH,
    price_path: str | Path = DEFAULT_PRICE_PATH,
    universe: Iterable[str],
    price_refresher: PriceRefresher,
    news_fetcher: NewsFetcher = fetch_live_news,
    now: datetime | None = None,
    research_config: ResearchConfig | None = None,
    lookback_days: int = 30,
    history_days: int = 90,
    top_n: int = 10,
) -> dict[str, object]:
    cycle_time = now or datetime.now(timezone.utc)
    if cycle_time.tzinfo is None or cycle_time.utcoffset() is None:
        raise ValueError("now must be timezone-aware")
    cycle_time = cycle_time.astimezone(timezone.utc)
    symbols = tuple(dict.fromkeys(str(symbol).strip().upper() for symbol in universe if str(symbol).strip()))
    if not symbols:
        raise ValueError("universe must contain at least one symbol")
    if lookback_days < 1 or history_days < lookback_days or top_n < 1:
        raise ValueError("history_days must cover a positive lookback and top_n must be positive")

    price_file = Path(price_path)
    price_file.parent.mkdir(parents=True, exist_ok=True)
    rows_added = price_refresher.refresh(
        price_file,
        symbols,
        start=cycle_time - timedelta(days=history_days),
        end=cycle_time,
    )
    rankings = volatility_rankings(price_file, limit=top_n)
    watchlist = [str(row["symbol"]) for row in rankings]
    if not watchlist:
        raise RuntimeError("no symbols have enough price history to rank")

    news_items = news_fetcher(db_path, symbols=watchlist)
    provider = CsvPriceProvider(price_file)
    config = replace(
        research_config or ResearchConfig(),
        symbols=tuple(watchlist),
        lookback_days=lookback_days,
    )
    research = run_research_cycle(
        db_path=db_path,
        price_provider=provider,
        config=config,
        now=cycle_time,
        return_summary=True,
    )
    resolved_outcomes = process_outcomes(db_path, provider, cycle_time)
    out_of_sample_evaluation = walk_forward_backtest(
        price_file,
        symbols=watchlist,
        horizon="1d",
        cost_bps=5.0,
    )

    return {
        "checked_at": cycle_time.isoformat(),
        "universe": list(symbols),
        "watchlist": watchlist,
        "price_rows_added": rows_added,
        "news_items_ingested": len(news_items),
        "signals_generated": research["signals_generated"],
        "signals": research["signals"],
        "strategy_ranking": research["strategy_ranking"],
        "strategy_drift": research["strategy_drift"],
        "out_of_sample_evaluation": out_of_sample_evaluation,
        "outcomes_resolved": resolved_outcomes,
    }


@dataclass
class AutonomousResearchScheduler:
    db_path: str | Path = DEFAULT_DB_PATH
    price_path: str | Path = DEFAULT_PRICE_PATH
    universe: tuple[str, ...] = ()
    price_refresher: PriceRefresher | None = None
    news_fetcher: NewsFetcher = fetch_live_news
    interval_seconds: float = 3600.0
    lookback_days: int = 30
    history_days: int = 90
    top_n: int = 10
    last_cycle_at: datetime | None = None

    def __post_init__(self) -> None:
        if self.price_refresher is None:
            self.price_refresher = FinnhubDailyPriceRefresher()
        if not self.universe:
            self.universe = default_research_symbols()
        if self.interval_seconds <= 0:
            raise ValueError("interval_seconds must be positive")

    def run_once(self, *, now: datetime | None = None) -> dict[str, object]:
        cycle_time = now or datetime.now(timezone.utc)
        result = run_autonomous_cycle(
            db_path=self.db_path,
            price_path=self.price_path,
            universe=self.universe,
            price_refresher=self.price_refresher,
            news_fetcher=self.news_fetcher,
            now=cycle_time,
            lookback_days=self.lookback_days,
            history_days=self.history_days,
            top_n=self.top_n,
        )
        self.last_cycle_at = cycle_time
        return result

    def run_if_due(self, *, now: datetime | None = None) -> dict[str, object] | None:
        cycle_time = now or datetime.now(timezone.utc)
        if self.last_cycle_at is not None:
            elapsed = (cycle_time - self.last_cycle_at).total_seconds()
            if elapsed < self.interval_seconds:
                return None
        return self.run_once(now=cycle_time)

    def run_forever(self) -> None:
        while True:
            started = time.monotonic()
            result = self.run_if_due()
            if result is not None:
                print(json.dumps(result, indent=2))
            elapsed = time.monotonic() - started
            time.sleep(max(1.0, self.interval_seconds - elapsed))


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Refresh market data and run a paper-trading research cycle.")
    parser.add_argument("--watch", action="store_true", help="repeat cycles at the configured interval")
    parser.add_argument("--interval", type=float, default=3600.0, help="seconds between cycles")
    parser.add_argument("--symbols", nargs="*", help="symbols to rank; defaults to RESEARCH_SYMBOLS")
    parser.add_argument("--top", type=int, default=10, help="number of volatile symbols to research")
    parser.add_argument("--lookback-days", type=int, default=30)
    parser.add_argument("--history-days", type=int, default=90)
    parser.add_argument("--db", type=Path, default=DEFAULT_DB_PATH)
    parser.add_argument("--prices", type=Path, default=DEFAULT_PRICE_PATH)
    args = parser.parse_args(argv)
    scheduler = AutonomousResearchScheduler(
        db_path=args.db,
        price_path=args.prices,
        universe=tuple(args.symbols) if args.symbols else default_research_symbols(),
        interval_seconds=args.interval,
        top_n=args.top,
        lookback_days=args.lookback_days,
        history_days=args.history_days,
    )
    if args.watch:
        scheduler.run_forever()
    else:
        print(json.dumps(scheduler.run_once(), indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

===== src/backtest.py =====

from __future__ import annotations

import csv
import math
import statistics
from collections import defaultdict
from datetime import datetime, timedelta, timezone
from pathlib import Path
from typing import Iterable

from .strategies import StrategyRegistry, get_default_registry


HORIZONS = {"1d": timedelta(days=1), "5d": timedelta(days=5), "20d": timedelta(days=20)}


def walk_forward_backtest(
    csv_path: str | Path,
    *,
    registry: StrategyRegistry | None = None,
    symbols: Iterable[str] | None = None,
    horizon: str = "1d",
    train_fraction: float = 0.7,
    cost_bps: float = 5.0,
) -> list[dict[str, object]]:
    """Evaluate fixed strategies on the chronological holdout portion of CSV history.

    Strategies receive only bars at or before each decision time. Costs are a
    round trip, applied as twice ``cost_bps`` to each directional trade.
    Results are per-signal diagnostics, not a portfolio simulation.
    """
    if horizon not in HORIZONS:
        raise ValueError("horizon must be one of 1d, 5d, or 20d")
    if not math.isfinite(train_fraction) or not 0 < train_fraction < 1:
        raise ValueError("train_fraction must be between 0 and 1")
    if not math.isfinite(cost_bps) or cost_bps < 0:
        raise ValueError("cost_bps must be finite and non-negative")

    selected_symbols = None if symbols is None else {str(symbol).upper() for symbol in symbols}
    histories: dict[str, list[tuple[datetime, float]]] = defaultdict(list)
    path = Path(csv_path)
    if not path.is_file():
        return []
    with path.open(newline="", encoding="utf-8") as csv_file:
        reader = csv.DictReader(csv_file)
        if not {"timestamp", "symbol", "price"} <= set(reader.fieldnames or ()):
            raise ValueError("CSV must contain timestamp, symbol, and price columns")
        for row in reader:
            symbol = row["symbol"].strip().upper()
            if not symbol or (selected_symbols is not None and symbol not in selected_symbols):
                continue
            try:
                timestamp = datetime.fromisoformat(row["timestamp"].replace("Z", "+00:00"))
                price = float(row["price"])
            except (TypeError, ValueError):
                continue
            if timestamp.tzinfo is None or not math.isfinite(price) or price <= 0:
                continue
            histories[symbol].append((timestamp.astimezone(timezone.utc), price))

    for history in histories.values():
        history.sort(key=lambda quote: quote[0])

    resolved_registry = registry or get_default_registry()
    results: dict[str, list[tuple[datetime, float, float]]] = defaultdict(list)
    round_trip_cost = 2 * cost_bps / 10_000
    holding_period = HORIZONS[horizon]

    for symbol, history in histories.items():
        if len(history) < 3:
            continue
        split_index = max(1, min(len(history) - 1, int(len(history) * train_fraction)))
        for strategy in resolved_registry.iterate():
            for index in range(split_index, len(history)):
                decision_time, entry_price = history[index]
                exit_quote = next(
                    (quote for quote in history[index + 1:] if quote[0] >= decision_time + holding_period),
                    None,
                )
                if exit_quote is None:
                    continue
                signal = strategy.generator(symbol, history[:index + 1], decision_time)
                if signal is None:
                    continue
                if signal.signal not in {"BUY", "SELL"}:
                    continue
                exit_price = exit_quote[1]
                gross_return = (
                    (exit_price - entry_price) / entry_price
                    if signal.signal == "BUY"
                    else (entry_price - exit_price) / entry_price
                )
                net_return = gross_return - round_trip_cost
                results[strategy.name].append((decision_time, gross_return, net_return))

    summaries: list[dict[str, object]] = []
    for strategy in resolved_registry.iterate():
        trades = sorted(results.get(strategy.name, ()), key=lambda row: row[0])
        if not trades:
            continue
        gross_returns = [row[1] for row in trades]
        net_returns = [row[2] for row in trades]
        summaries.append(
            {
                "model_name": strategy.name,
                "out_of_sample_start": trades[0][0].isoformat(),
                "trades": len(trades),
                "horizon": horizon,
                "round_trip_cost_bps": 2 * cost_bps,
                "average_gross_return": statistics.mean(gross_returns),
                "average_net_return": statistics.mean(net_returns),
                "median_net_return": statistics.median(net_returns),
                "win_rate": sum(value > 0 for value in net_returns) / len(net_returns),
            }
        )
    return summaries

from __future__ import annotations

import math
import sqlite3
from dataclasses import dataclass
from datetime import date, datetime, timezone
from pathlib import Path
from typing import Callable, Protocol

from .prices import PriceProvider
from .risk import AccountState


@dataclass
class SimOrder:
    client_order_id: str
    symbol: str
    side: str
    qty: int
    fill_price: float | None
    status: str
    created_at: datetime


class Broker(Protocol):
    def get_account(self) -> AccountState: ...

    def get_positions(self) -> dict[str, int]: ...

    def submit_order(
        self, symbol: str, side: str, qty: int, client_order_id: str
    ) -> SimOrder: ...

    def get_order(self, client_order_id: str) -> SimOrder | None: ...


def _utc_now() -> datetime:
    return datetime.now(timezone.utc)


class SimBroker:
    def __init__(
        self,
        price_provider: PriceProvider,
        *,
        slippage: float = 0.001,
        max_quote_age_seconds: int = 60,
        state_db_path: str | Path | None = None,
        clock: Callable[[], datetime] = _utc_now,
    ) -> None:
        if not math.isfinite(slippage) or not 0 <= slippage < 1:
            raise ValueError("slippage must be finite and in [0, 1)")
        if (
            not isinstance(max_quote_age_seconds, int)
            or isinstance(max_quote_age_seconds, bool)
            or max_quote_age_seconds < 0
        ):
            raise ValueError("max_quote_age_seconds must be a non-negative integer")
        self._price_provider = price_provider
        self._slippage = slippage
        self._max_quote_age_seconds = max_quote_age_seconds
        self._clock = clock
        self._state_db_path = Path(state_db_path) if state_db_path is not None else None
        self._cash = 10_000.0
        self._day_start_equity = 10_000.0
        self._day_start_equity_date: date | None = None
        self._positions: dict[str, int] = {}
        self._orders: dict[str, SimOrder] = {}
        if self._state_db_path is not None:
            self._load_state()

    def _now(self) -> datetime:
        now = self._clock()
        if now.tzinfo is None or now.utcoffset() is None:
            raise ValueError("broker clock must return a timezone-aware datetime")
        return now.astimezone(timezone.utc)

    def get_account(self) -> AccountState:
        now = self._now()
        positions: dict[str, float] = {}
        for symbol, qty in self._positions.items():
            price = self._price_provider.get_price(symbol, now)
            if price is None or not math.isfinite(price) or price <= 0:
                raise ValueError(f"no valid price available for held symbol {symbol}")
            positions[symbol] = qty * price
        equity = self._cash + sum(positions.values())
        current_day = now.date()
        if self._day_start_equity_date != current_day:
            self._day_start_equity = equity
            self._day_start_equity_date = current_day
            self._persist_state()
        return AccountState(
            equity=equity,
            day_start_equity=self._day_start_equity,
            cash=self._cash,
            positions=positions,
            as_of=now,
        )

    def get_positions(self) -> dict[str, int]:
        return dict(self._positions)

    def get_order(self, client_order_id: str) -> SimOrder | None:
        return self._orders.get(client_order_id)

    def submit_order(
        self,
        symbol: str,
        side: str,
        qty: int,
        client_order_id: str,
    ) -> SimOrder:
        existing = self.get_order(client_order_id)
        if existing is not None:
            return existing
        now = self._now()

        def reject() -> SimOrder:
            order = SimOrder(client_order_id, symbol, side, qty, None, "rejected", now)
            self._orders[client_order_id] = order
            self._persist_state()
            return order

        if not isinstance(qty, int) or isinstance(qty, bool) or qty <= 0:
            return reject()
        if side not in ("BUY", "SELL"):
            return reject()
        if side == "SELL" and qty > self._positions.get(symbol, 0):
            return reject()

        quote = self._price_provider.get_quote(symbol, now)
        if quote is None:
            return reject()
        quote_age_seconds = (now - quote.timestamp.astimezone(timezone.utc)).total_seconds()
        if (
            quote.timestamp.tzinfo is None
            or quote.timestamp.utcoffset() is None
            or quote_age_seconds < 0
            or quote_age_seconds > self._max_quote_age_seconds
            or not math.isfinite(quote.price)
            or quote.price <= 0
        ):
            return reject()
        market_price = quote.price
        fill_price = market_price * (1 + self._slippage if side == "BUY" else 1 - self._slippage)
        if side == "BUY" and qty * fill_price > self._cash:
            return reject()

        if side == "BUY":
            self._cash -= qty * fill_price
            self._positions[symbol] = self._positions.get(symbol, 0) + qty
        else:
            self._cash += qty * fill_price
            remaining = self._positions[symbol] - qty
            if remaining:
                self._positions[symbol] = remaining
            else:
                del self._positions[symbol]

        order = SimOrder(client_order_id, symbol, side, qty, fill_price, "filled", now)
        self._orders[client_order_id] = order
        self._persist_state()
        return order

    def _load_state(self) -> None:
        assert self._state_db_path is not None
        self._state_db_path.parent.mkdir(parents=True, exist_ok=True)
        with sqlite3.connect(self._state_db_path) as connection:
            connection.executescript(
                """
                CREATE TABLE IF NOT EXISTS broker_account (
                    id INTEGER PRIMARY KEY CHECK (id = 1),
                    cash REAL NOT NULL,
                    day_start_equity REAL NOT NULL,
                    day_start_equity_date TEXT
                );
                CREATE TABLE IF NOT EXISTS broker_positions (
                    symbol TEXT PRIMARY KEY,
                    qty INTEGER NOT NULL
                );
                CREATE TABLE IF NOT EXISTS broker_orders (
                    client_order_id TEXT PRIMARY KEY,
                    symbol TEXT NOT NULL,
                    side TEXT NOT NULL,
                    qty INTEGER NOT NULL,
                    fill_price REAL,
                    status TEXT NOT NULL,
                    created_at TEXT NOT NULL
                );
                """
            )
            account = connection.execute(
                "SELECT cash, day_start_equity, day_start_equity_date FROM broker_account WHERE id = 1"
            ).fetchone()
            if account is not None:
                self._cash = float(account[0])
                self._day_start_equity = float(account[1])
                self._day_start_equity_date = date.fromisoformat(account[2]) if account[2] else None
            self._positions = dict(connection.execute("SELECT symbol, qty FROM broker_positions"))
            for row in connection.execute(
                "SELECT client_order_id, symbol, side, qty, fill_price, status, created_at FROM broker_orders"
            ):
                order = SimOrder(
                    row[0], row[1], row[2], int(row[3]), row[4], row[5], datetime.fromisoformat(row[6])
                )
                self._orders[order.client_order_id] = order
        self._persist_state()

    def _persist_state(self) -> None:
        if self._state_db_path is None:
            return
        with sqlite3.connect(self._state_db_path) as connection:
            with connection:
                connection.execute(
                    "INSERT INTO broker_account (id, cash, day_start_equity, day_start_equity_date) "
                    "VALUES (1, ?, ?, ?) ON CONFLICT(id) DO UPDATE SET "
                    "cash=excluded.cash, day_start_equity=excluded.day_start_equity, "
                    "day_start_equity_date=excluded.day_start_equity_date",
                    (
                        self._cash,
                        self._day_start_equity,
                        self._day_start_equity_date.isoformat() if self._day_start_equity_date else None,
                    ),
                )
                connection.execute("DELETE FROM broker_positions")
                connection.executemany(
                    "INSERT INTO broker_positions (symbol, qty) VALUES (?, ?)",
                    self._positions.items(),
                )
                connection.execute("DELETE FROM broker_orders")
                connection.executemany(
                    "INSERT INTO broker_orders "
                    "(client_order_id, symbol, side, qty, fill_price, status, created_at) "
                    "VALUES (?, ?, ?, ?, ?, ?, ?)",
                    (
                        (
                            order.client_order_id,
                            order.symbol,
                            order.side,
                            order.qty,
                            order.fill_price,
                            order.status,
                            order.created_at.isoformat(),
                        )
                        for order in self._orders.values()
                    ),
                )

@contextmanager
def _readonly_connection(db_path: str | Path) -> Iterator[sqlite3.Connection]:
    uri = Path(db_path).resolve().as_uri() + "?mode=ro"
    with closing(sqlite3.connect(uri, uri=True)) as connection:
        connection.row_factory = sqlite3.Row
        yield connection


def _query_frame(
    db_path: str | Path,
    query: str,
    columns: list[str],
    params: tuple[object, ...] = (),
) -> pd.DataFrame:
    if not Path(db_path).is_file():
        return pd.DataFrame(columns=columns)
    try:
        with _readonly_connection(db_path) as connection:
            return pd.read_sql_query(query, connection, params=params)
    except sqlite3.OperationalError as error:
        if "no such table" in str(error).lower():
            return pd.DataFrame(columns=columns)
        raise

def create_tables(db_path: DatabasePath) -> None:
    with closing(_connect(db_path)) as connection:
        with connection:
            connection.executescript(
                """

                    id INTEGER PRIMARY KEY,
                    source TEXT NOT NULL,
                    fetched_at TEXT NOT NULL,
                    text TEXT NOT NULL,
                    title TEXT,
                    url TEXT,
                    published_at TEXT
                );

                CREATE TABLE IF NOT EXISTS news_events (
                    id INTEGER PRIMARY KEY,
                    item_id INTEGER NOT NULL,
                    symbol TEXT NOT NULL,
                    title TEXT NOT NULL,
                    summary TEXT NOT NULL,
                    event_type TEXT NOT NULL,
                    sentiment REAL NOT NULL CHECK (sentiment BETWEEN 0 AND 1),
                    relevance_score REAL NOT NULL CHECK (relevance_score BETWEEN 0 AND 1),
                    quality_score REAL NOT NULL CHECK (quality_score BETWEEN 0 AND 1),
                    source TEXT NOT NULL,
                    published_at TEXT,
                    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                    FOREIGN KEY (item_id) REFERENCES raw_items (id)
                );

                CREATE TABLE IF NOT EXISTS signals (
                    id INTEGER PRIMARY KEY,
                    item_id INTEGER,
                    symbol TEXT NOT NULL,
                    signal TEXT NOT NULL CHECK (signal IN ('BUY', 'SELL', 'HOLD')),
                    confidence REAL NOT NULL CHECK (confidence BETWEEN 0 AND 1),
                    horizon TEXT NOT NULL CHECK (horizon IN ('1d', '5d', '20d')),
                    rationale TEXT NOT NULL,
                    model_name TEXT NOT NULL,
                    prompt_version TEXT NOT NULL,
                    price_at_signal REAL,
                    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                    FOREIGN KEY (item_id) REFERENCES raw_items (id)
                );

                CREATE TABLE IF NOT EXISTS outcomes (
                    id INTEGER PRIMARY KEY,
                    signal_id INTEGER NOT NULL,
                    horizon_checked TEXT NOT NULL,
                    price_at_check REAL NOT NULL,
                    return_pct REAL NOT NULL,
                    no_trade INTEGER NOT NULL DEFAULT 0 CHECK (no_trade IN (0, 1)),
                    checked_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,

                    id INTEGER PRIMARY KEY,
                    symbol TEXT,
                    side TEXT,
                    price REAL,
                    confidence REAL,
                    price_timestamp TEXT,

                );

                CREATE TABLE IF NOT EXISTS daily_reports (
                    date TEXT PRIMARY KEY,
                    text TEXT NOT NULL
                );
                """
            )
            outcome_columns = {
                row["name"] for row in connection.execute("PRAGMA table_info(outcomes)")
            }
            if "no_trade" not in outcome_columns:
                connection.execute(
                    "ALTER TABLE outcomes ADD COLUMN no_trade INTEGER NOT NULL "
                    "DEFAULT 0 CHECK (no_trade IN (0, 1))"
                )

            raw_columns = {row["name"] for row in connection.execute("PRAGMA table_info(raw_items)")}
            for column_name, column_type in {
                "title": "TEXT",
                "url": "TEXT",
                "published_at": "TEXT",
            }.items():
                if column_name not in raw_columns:
                    connection.execute(
                        f"ALTER TABLE raw_items ADD COLUMN {column_name} {column_type}"
                    )

def log_risk_decision(
    db_path: DatabasePath,
    trade: ProposedTrade,
    decision: RiskDecision,
    *,
    created_at: str | None = None,
) -> int:
    def audit_value(value: object) -> object:
        if value is None:
            return None
        if isinstance(value, datetime):
            return value.isoformat()
        if isinstance(value, float) and not math.isfinite(value):
            return repr(value)
        if isinstance(value, (str, int, float)):
            return value
        return str(value)

        with connection:
            cursor = connection.execute(
                """
                INSERT INTO risk_events (
                    symbol, side, price, confidence, price_timestamp,
                    approved, qty, reasons, created_at

                """,
                (
                    audit_value(trade.symbol),
                    audit_value(trade.side),
                    audit_value(trade.price),
                    audit_value(trade.confidence),
                    audit_value(trade.price_timestamp),
                    int(decision.approved),
                    decision.qty,
                    json.dumps(decision.reasons),
                    created_at,
                ),
            )
            return int(cursor.lastrowid)

from __future__ import annotations

import logging

from .broker import Broker, SimOrder

logger = logging.getLogger(__name__)
KILL_SWITCH_PATH = Path(__file__).resolve().parents[1] / "KILL"


def _log_order(db_path: DatabasePath, signal_id: int, order: SimOrder) -> None:
    log_order(
        db_path,
        client_order_id=order.client_order_id,
        signal_id=signal_id,
        symbol=order.symbol,
        side=order.side,
        qty=order.qty,
        fill_price=order.fill_price,
        status=order.status,
        created_at=order.created_at.isoformat(),
    )


def process_signal(
    signal: Mapping[str, object],
    price: float,
    broker: Broker,
    config: RiskConfig,
    now: datetime,
    *,
    price_timestamp: datetime,
    db_path: DatabasePath = "data/research.db",
) -> SimOrder | None:
    try:
        signal_id = signal["id"]
        if not isinstance(signal_id, int) or isinstance(signal_id, bool):
            raise ValueError("signal id must be an integer")
        symbol = signal["symbol"]
        signal_type = signal["signal"]
        confidence = signal["confidence"]
        if not isinstance(symbol, str) or not isinstance(signal_type, str):
            raise ValueError("signal symbol and type must be strings")
        if signal_type == "HOLD":
            logger.info("HOLD signal %s logged; no order submitted", signal_id)
            return None

        trade = ProposedTrade(symbol, signal_type, price, confidence, price_timestamp)
        create_tables(db_path)
        client_order_id = f"signal-{signal_id}"
        if KILL_SWITCH_PATH.exists():
            decision = RiskDecision(False, 0, ["KILL file exists; order submission disabled"])
            log_risk_decision(db_path, trade, decision)
            logger.warning("Signal %s blocked by KILL switch", signal_id)
            return None

        existing = broker.get_order(client_order_id)
        if existing is not None:
            _log_order(db_path, signal_id, existing)
            return existing

        decision = check_trade(trade, broker.get_account(), config, now)
        log_risk_decision(db_path, trade, decision)
        if not decision.approved:
            logger.info("Signal %s rejected: %s", signal_id, "; ".join(decision.reasons))
            return None

        order = broker.submit_order(symbol, signal_type, decision.qty, client_order_id)
        _log_order(db_path, signal_id, order)
        logger.info(
            "Order %s %s %s qty=%s status=%s",
            order.client_order_id,
            order.side,
            order.symbol,
            order.qty,
            order.status,
        )
        return order
    except Exception:
        logger.exception("Signal processing failed; no order submitted")
        return None

def volatility_rankings(csv_path: str | Path = DEFAULT_PRICE_PATH, *, limit: int = 10) -> list[dict[str, object]]:
    if limit < 1:
        raise ValueError("limit must be positive")
    path = Path(csv_path)
    if not path.is_file():
        return []

    prices_by_symbol: dict[str, list[tuple[datetime, float]]] = {}
    with path.open(newline="", encoding="utf-8") as csv_file:
        reader = csv.DictReader(csv_file)
        if not {"timestamp", "symbol", "price"} <= set(reader.fieldnames or ()):
            return []
        for row in reader:
            try:
                timestamp = datetime.fromisoformat(row["timestamp"].replace("Z", "+00:00"))
                price = float(row["price"])
            except (TypeError, ValueError):
                continue
            symbol = row["symbol"].strip().upper()
            if not symbol or timestamp.tzinfo is None or not math.isfinite(price) or price <= 0:
                continue
            prices_by_symbol.setdefault(symbol, []).append((timestamp.astimezone(timezone.utc), price))

    rankings = []
    for symbol, history in prices_by_symbol.items():
        history.sort(key=lambda item: item[0])
        returns = [current / previous - 1 for (_, previous), (_, current) in zip(history, history[1:])]
        if not returns:
            continue
        absolute_returns = [abs(value) for value in returns]
        rankings.append(
            {
                "symbol": symbol,
                "latest_price": history[-1][1],
                "daily_return_std": statistics.stdev(returns) if len(returns) > 1 else 0.0,
                "average_absolute_return": statistics.mean(absolute_returns),
                "max_absolute_return": max(absolute_returns),
                "days_seen": len(history),
            }
        )
    rankings.sort(
        key=lambda item: (
            item["daily_return_std"],
            item["average_absolute_return"],
            item["max_absolute_return"],
        ),
        reverse=True,
    )
    return rankings[:limit]

        self._last_request = 0.0

    def refresh(
        self,
        csv_path: str | Path,
        symbols: Iterable[str],
        *,
        start: datetime,
        end: datetime,
    ) -> int:
        if not self.api_key:
            raise ValueError("FINNHUB_API_KEY is required to refresh daily prices")
        if start.tzinfo is None or start.utcoffset() is None:
            raise ValueError("start must be timezone-aware")
        if end.tzinfo is None or end.utcoffset() is None:
            raise ValueError("end must be timezone-aware")
        start_utc = start.astimezone(timezone.utc)
        end_utc = end.astimezone(timezone.utc)
        if start_utc >= end_utc:
            raise ValueError("start must be earlier than end")

        path = Path(csv_path)
        path.parent.mkdir(parents=True, exist_ok=True)
        prices: dict[tuple[str, str], float] = {}
        if path.exists():
            with path.open(newline="", encoding="utf-8") as csv_file:
                reader = csv.DictReader(csv_file)
                if not {"timestamp", "symbol", "price"} <= set(reader.fieldnames or ()):
                    raise ValueError("CSV must contain timestamp, symbol, and price columns")
                for row in reader:
                    stamp = self._normalize_timestamp(row["timestamp"])
                    symbol = row["symbol"].strip().upper()
                    price = float(row["price"])
                    if symbol and math.isfinite(price) and price > 0:
                        prices[(stamp, symbol)] = price

        added = 0
        for symbol in dict.fromkeys(str(value).strip().upper() for value in symbols if str(value).strip()):
            query = parse.urlencode(
                {
                    "symbol": symbol,
                    "resolution": "D",
                    "from": int(start_utc.timestamp()),
                    "to": int(end_utc.timestamp()),
                    "token": self.api_key,
                }
            )
            req = request.Request(f"https://finnhub.io/api/v1/stock/candle?{query}")
            try:
                elapsed = time.monotonic() - self._last_request
                if self._last_request and elapsed < self.request_interval_seconds:
                    time.sleep(self.request_interval_seconds - elapsed)
                self._last_request = time.monotonic()
                with request.urlopen(req, timeout=self.timeout) as response:
                    payload = json.loads(response.read().decode("utf-8"))
            except (error.URLError, TimeoutError, ValueError) as exc:
                raise RuntimeError(f"Finnhub daily candles failed for {symbol}") from exc
            if not isinstance(payload, dict) or payload.get("s") != "ok":
                continue
            timestamps = payload.get("t", [])
            closes = payload.get("c", [])
            for timestamp, close in zip(timestamps, closes):
                try:
                    price = float(close)
                    quote_time = datetime.fromtimestamp(float(timestamp), tz=timezone.utc)
                except (TypeError, ValueError, OverflowError, OSError):
                    continue
                if not math.isfinite(price) or price <= 0 or quote_time > end_utc:
                    continue
                stamp = quote_time.isoformat().replace("+00:00", "Z")
                key = (stamp, symbol)
                if key not in prices:
                    added += 1
                prices[key] = price

        self._write_csv(path, prices)
        return added

    @staticmethod
    def _normalize_timestamp(value: str) -> str:
        text = value.strip()
        if text.endswith("Z"):
            text = text[:-1] + "+00:00"
        parsed = datetime.fromisoformat(text)
        if parsed.tzinfo is None or parsed.utcoffset() is None:
            raise ValueError("price timestamps must be timezone-aware")
        return parsed.astimezone(timezone.utc).isoformat().replace("+00:00", "Z")

    @staticmethod
    def _write_csv(path: Path, prices: dict[tuple[str, str], float]) -> None:
        with tempfile.NamedTemporaryFile(
            "w", newline="", encoding="utf-8", dir=path.parent, delete=False
        ) as csv_file:
            temporary_path = Path(csv_file.name)
            writer = csv.writer(csv_file)
            writer.writerow(("timestamp", "symbol", "price"))
            for (timestamp, symbol), price in sorted(prices.items(), key=lambda item: (item[0][1], item[0][0])):
                writer.writerow((timestamp, symbol, f"{price:.8f}"))
        temporary_path.replace(path)


class FinnhubQuoteProvider:
    def __init__(self, api_key: str | None = None, *, timeout: float = 15) -> None:
        self.api_key = api_key or os.getenv("FINNHUB_API_KEY")
        self.timeout = timeout

    def get_quote(self, symbol: str, when: datetime) -> Quote | None:
        if when.tzinfo is None or when.utcoffset() is None:
            raise ValueError("when must be timezone-aware")
        if not self.api_key:
            return None
        query = parse.urlencode({"symbol": symbol.upper(), "token": self.api_key})
        req = request.Request(f"https://finnhub.io/api/v1/quote?{query}")
        try:
            with request.urlopen(req, timeout=self.timeout) as response:
                payload = json.loads(response.read().decode("utf-8"))
            price = float(payload["c"])
            timestamp = datetime.fromtimestamp(float(payload["t"]), tz=timezone.utc)
        except (error.URLError, TimeoutError, ValueError, TypeError, KeyError, OverflowError, OSError):
            return None
        if not math.isfinite(price) or price <= 0 or timestamp > when.astimezone(timezone.utc):
            return None
        return Quote(price, timestamp)

    def get_price(self, symbol: str, when: datetime) -> float | None:
        quote = self.get_quote(symbol, when)
        return None if quote is None else quote.price

def _normalize_datetime(value: str | None) -> str:
    if value is None or not value.strip():
        return datetime.now(timezone.utc).isoformat()
    text = value.strip()
    if text.endswith("Z"):
        text = text[:-1] + "+00:00"
    try:
        dt = datetime.fromisoformat(text)
    except ValueError:
        try:
            timestamp = float(text)
            if abs(timestamp) >= 1_000_000_000_000:
                timestamp /= 1000.0
            return datetime.fromtimestamp(timestamp, timezone.utc).isoformat()
        except (ValueError, OverflowError, OSError):
            raise ValueError(f"invalid news publication timestamp: {value!r}") from None
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return dt.astimezone(timezone.utc).isoformat()


def _dedupe_key(article: dict[str, object]) -> str:
    href = str(article.get("url") or "").strip()
    if href:
        return "url:" + href
    title = str(article.get("title") or "").strip()
    content = str(article.get("content") or "").strip()
    payload = (title + "\n" + content).strip()
    return "content:" + hashlib.sha256(payload.encode("utf-8")).hexdigest()


def _detect_event_type(title: str, content: str) -> str:
    text = (title + " " + content).lower()
    if "guidance" in text or "raises guidance" in text or "guidance upgrade" in text:
        return "guidance"
    if "earnings" in text or "revenue" in text or "profit" in text:
        return "earnings"
    if "merger" in text or "acquisition" in text or "deal" in text:
        return "deal"
    if "regulatory" in text or "sec" in text or "compliance" in text:
        return "regulatory"
    if "launch" in text or "product" in text:
        return "product"
    return "general"


def _sentiment_score(title: str, content: str) -> float:
    text = (title + " " + content).lower()
    positives = [
        "strong",
        "beat",
        "raise",
        "rises",
        "growth",
        "upbeat",
        "surge",
        "outperform",
        "boost",
        "guidance",
        "expands",
    ]
    negatives = [
        "miss",
        "weak",
        "cut",
        "lower",
        "downgrade",
        "drop",
        "decline",
        "down",
        "issue",
        "delay",
    ]
    positive_hits = sum(1 for term in positives if term in text)
    negative_hits = sum(1 for term in negatives if term in text)
    score = (positive_hits - negative_hits) / max(1, positive_hits + negative_hits + 1)
    return max(0.0, min(1.0, (score + 1.0) / 2.0))


def normalize_news_item(article: dict[str, object]) -> dict[str, object]:
    title = str(article.get("title") or article.get("headline") or "Untitled").strip()
    content = str(article.get("content") or article.get("text") or title).strip()
    symbols = article.get("symbols") or []
    if isinstance(symbols, str):
        symbols = [symbols]
    symbol_list = [str(symbol).upper().strip() for symbol in list(symbols)]
    symbol_list = [symbol for symbol in symbol_list if symbol]
    if not symbol_list:
        symbol_list = ["UNKNOWN"]

    event_type = _detect_event_type(title, content)
    sentiment = _sentiment_score(title, content)
    relevance = min(1.0, 0.55 + (0.1 * len(symbol_list)) + (0.15 if event_type != "general" else 0.0))
    quality = min(1.0, 0.65 + (0.15 if article.get("source") else 0.0))
    summary = content[:280] if content else title

    published_at = article.get("published_at") or article.get("published")
    return {
        "source": str(article.get("source") or "unknown"),
        "title": title,
        "content": content,
        "summary": summary,
        "published_at": _normalize_datetime(None if published_at is None else str(published_at)),
        "url": str(article.get("url") or article.get("link") or ""),
        "symbols": symbol_list,
        "event_type": event_type,
        "sentiment": float(sentiment),
        "relevance_score": float(relevance),
        "quality_score": float(quality),
    }


def ingest_news_articles(
    db_path: DatabasePath,
    articles: Iterable[dict[str, object]],
) -> list[dict[str, object]]:
    path = Path(db_path)
    path.parent.mkdir(parents=True, exist_ok=True)
    create_tables(path)

    seen_keys = {
        _dedupe_key(
            {
                "url": row.get("url"),
                "title": row.get("title"),
                "content": row.get("text"),
            }
        )
        for row in fetch_news_items(path)
    }

    inserted: list[dict[str, object]] = []
    for article in articles:
        normalized = normalize_news_item(article)
        dedupe_key = _dedupe_key(normalized)
        if dedupe_key in seen_keys:
            continue

        item_id = insert_raw_item(
            path,
            source=normalized["source"],
            fetched_at=_normalize_datetime(None),
            text=normalized["content"],
            title=normalized["title"],
            url=normalized["url"],
            published_at=normalized["published_at"],
        )
        for symbol in normalized["symbols"]:
            event_id = insert_news_event(
                path,
                item_id=item_id,
                symbol=str(symbol),
                title=str(normalized["title"]),
                summary=str(normalized["summary"]),
                event_type=str(normalized["event_type"]),
                sentiment=float(normalized["sentiment"]),
                relevance_score=float(normalized["relevance_score"]),
                quality_score=float(normalized["quality_score"]),
                source=str(normalized["source"]),
                published_at=str(normalized["published_at"]),
            )
            inserted.append(
                {
                    "id": event_id,
                    "item_id": item_id,
                    "symbol": symbol,
                    "title": normalized["title"],
                    "summary": normalized["summary"],
                    "event_type": normalized["event_type"],
                    "sentiment": normalized["sentiment"],
                    "relevance_score": normalized["relevance_score"],
                    "quality_score": normalized["quality_score"],
                    "source": normalized["source"],
                    "published_at": normalized["published_at"],
                }
            )
        seen_keys.add(dedupe_key)
    return inserted


def fetch_news_items(db_path: DatabasePath, *, limit: int | None = None) -> list[dict[str, object]]:
    query = "SELECT * FROM raw_items ORDER BY id DESC"
    if limit is not None:
        query += " LIMIT ?"
    connection = connect(db_path)
    connection.row_factory = Row
    try:
        rows = connection.execute(query, ((limit,) if limit is not None else ())).fetchall()
    finally:
        connection.close()
    return [dict(row) for row in rows]

class NewsSource(ABC):
    @abstractmethod
    def fetch_articles(self) -> list[dict[str, object]]:
        ...


def normalize_news_article(item: dict[str, object], *, default_source: str = "unknown") -> dict[str, object]:
    if not isinstance(item, dict):
        raise TypeError("expected a dict-like article payload")

    normalized = dict(item)
    source = str(normalized.get("source") or normalized.get("provider") or default_source)
    title = str(normalized.get("title") or normalized.get("headline") or "Untitled").strip()
    content = str(normalized.get("content") or normalized.get("text") or title).strip()
    symbols = normalized.get("symbols") or normalized.get("symbol") or []
    if isinstance(symbols, str):
        list_symbols = [symbols]
    elif isinstance(symbols, (list, tuple, set)):
        list_symbols = list(symbols)
    else:
        list_symbols = []
    normalized_symbols = [str(symbol).upper().strip() for symbol in list_symbols if str(symbol).strip()]
    if not normalized_symbols and normalized.get("symbol"):
        normalized_symbols = [str(normalized["symbol"]).upper().strip()]
    if not normalized_symbols:
        normalized_symbols = ["UNKNOWN"]

    published_at = normalized.get("published_at") or normalized.get("published") or normalized.get("datetime")
    if published_at is None:
        published_at = datetime.now(timezone.utc).isoformat()
    else:
        published_text = str(published_at).strip()
        if published_text.endswith("Z"):
            published_text = published_text[:-1] + "+00:00"
        try:
            published_dt = datetime.fromisoformat(published_text)
            if published_dt.tzinfo is None:
                published_dt = published_dt.replace(tzinfo=timezone.utc)
            published_at = published_dt.astimezone(timezone.utc).isoformat()
        except ValueError:
            published_at = published_text

    normalized["source"] = source
    normalized["title"] = title
    normalized["content"] = content
    normalized["published_at"] = published_at
    normalized["url"] = str(normalized.get("url") or normalized.get("link") or "")
    normalized["symbols"] = normalized_symbols
    normalized["symbol"] = normalized_symbols[0]
    return normalized


class JsonFeedSource(NewsSource):
    def __init__(self, path: str | Path) -> None:
        self.path = Path(path)

    def fetch_articles(self) -> list[dict[str, object]]:
        if not self.path.exists():
            return []
        payload = json.loads(self.path.read_text(encoding="utf-8"))
        if isinstance(payload, dict):
            payload = payload.get("articles", [])
        items: list[dict[str, object]] = []
        for item in payload:
            if not isinstance(item, dict):
                continue
            items.append(normalize_news_article(item, default_source=self.path.stem))
        return items


class FinnhubNewsSource(NewsSource):
    def __init__(self, api_key: str | None = None, symbols: Iterable[str] | None = None) -> None:
        self.api_key = api_key or os.getenv("FINNHUB_API_KEY")
        self.symbols = tuple(symbols or ())

    def fetch_articles(self) -> list[dict[str, object]]:
        if not self.api_key:
            return []
        url = "https://finnhub.io/api/v1/news?category=general"
        req = request.Request(url, headers={"X-Finnhub-Token": self.api_key})
        try:
            with request.urlopen(req, timeout=15) as response:
                payload = json.loads(response.read().decode("utf-8"))
        except (error.URLError, ValueError, TimeoutError):
            return []

        for article in payload:

                normalized["symbol"] = normalized["symbols"][0]

        return items


_IMPORTANT_8K_ITEMS = {"1.03", "1.05", "2.02", "2.04", "2.06", "3.01", "4.02", "5.01", "5.02"}
_8K_ITEM_LABELS = {
    "1.03": "bankruptcy",
    "1.05": "cybersecurity incident",
    "2.02": "earnings results",
    "2.04": "debt default or acceleration",
    "2.06": "impairment",
    "3.01": "delisting notice",
    "4.02": "past financials no longer reliable",
    "5.01": "change in control",
    "5.02": "officer or director change",
}


def _feed_datetime(value: str) -> str | None:
    text = value.strip()
    if not text:
        return None
    try:
        parsed = email.utils.parsedate_to_datetime(text)
    except (TypeError, ValueError, IndexError):
        try:
            parsed = datetime.fromisoformat(text.replace("Z", "+00:00"))
        except ValueError:
            return None
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=timezone.utc)
    return parsed.astimezone(timezone.utc).isoformat()


def parse_yahoo_rss(payload: bytes | str, symbol: str) -> list[dict[str, object]]:
    root = ElementTree.fromstring(payload)
    articles: list[dict[str, object]] = []
    for item in root.iter("item"):
        def text(name: str) -> str:
            element = item.find(name)
            return "" if element is None or element.text is None else element.text.strip()

        title = text("title")
        if not title:
            continue
        published_at = _feed_datetime(text("pubDate"))
        articles.append(
            {
                "source": "Yahoo Finance",
                "title": title,
                "content": text("description") or title,
                "published_at": published_at,
                "url": text("link"),
                "symbols": [symbol.upper()],
                "symbol": symbol.upper(),
                "external_id": text("guid"),
                "kind": "headline",
            }
        )
    return articles


def parse_sec_atom(payload: bytes | str, symbol: str) -> list[dict[str, object]]:
    root = ElementTree.fromstring(payload)

    def local_name(tag: str) -> str:
        return tag.rsplit("}", 1)[-1]

    articles: list[dict[str, object]] = []
    for entry in root.iter():
        if local_name(entry.tag) != "entry":
            continue
        fields = {
            local_name(element.tag): (element.text or "").strip()
            for element in entry.iter()
            if element is not entry
        }
        form = fields.get("filing-type", "filing")
        item_codes = sorted(set(re.findall(r"\d+\.\d+", fields.get("items-desc", ""))))
        labels = [_8K_ITEM_LABELS.get(code, f"item {code}") for code in item_codes]
        title = f"{symbol.upper()} filed a {form}"
        if labels:
            title += ": " + ", ".join(labels)
        important = bool(_IMPORTANT_8K_ITEMS.intersection(item_codes))
        if important:
            title = "! " + title
        link = next(
            (element.get("href", "") for element in entry.iter() if local_name(element.tag) == "link"),
            "",
        )
        articles.append(
            {
                "source": "SEC EDGAR",
                "title": title,
                "content": fields.get("form-name") or title,
                "published_at": _feed_datetime(fields.get("updated", "")),
                "url": link,
                "symbols": [symbol.upper()],
                "symbol": symbol.upper(),
                "external_id": fields.get("accession-number", ""),
                "kind": "filing",
                "form": form,
                "items": item_codes,
                "important": important,
            }
        )
    return articles


class PublicFeedNewsSource(NewsSource):
    def __init__(
        self,
        symbols: Iterable[str],
        *,
        sec_user_agent: str | None = None,
        timeout: float = 15,
    ) -> None:
        self.symbols = tuple(dict.fromkeys(str(symbol).strip().upper() for symbol in symbols if str(symbol).strip()))
        candidate_user_agent = sec_user_agent or os.getenv("SEC_USER_AGENT")
        self.sec_user_agent = (
            candidate_user_agent
            if candidate_user_agent and "@" in candidate_user_agent and "you@example.com" not in candidate_user_agent
            else None
        )
        self.timeout = timeout
        self._last_sec_request = 0.0

    def _fetch(self, url: str, *, user_agent: str) -> bytes:
        req = request.Request(url, headers={"User-Agent": user_agent})
        with request.urlopen(req, timeout=self.timeout) as response:
            return response.read()

    def _fetch_sec_url(self, url: str) -> bytes:
        if self._last_sec_request:
            elapsed = time.monotonic() - self._last_sec_request
            if elapsed < 0.15:
                time.sleep(0.15 - elapsed)
        self._last_sec_request = time.monotonic()
        return self._fetch(url, user_agent=self.sec_user_agent or "")

    def _fetch_sec(self, symbol: str, cik: str) -> bytes:
        query = urlencode(
            {
                "action": "getcompany",
                "CIK": cik,
                "type": "8-K",
                "dateb": "",
                "owner": "include",
                "count": 10,
                "output": "atom",
            }
        )
        return self._fetch_sec_url("https://www.sec.gov/cgi-bin/browse-edgar?" + query)

    def fetch_articles(self) -> list[dict[str, object]]:
        articles: list[dict[str, object]] = []
        cik_by_symbol: dict[str, str] | None = None
        for symbol in self.symbols:
            yahoo_url = "https://feeds.finance.yahoo.com/rss/2.0/headline?" + urlencode(
                {"s": symbol, "region": "US", "lang": "en-US"}
            )
            try:
                articles.extend(
                    parse_yahoo_rss(
                        self._fetch(yahoo_url, user_agent="QuantResearch/1.0"),
                        symbol,
                    )
                )
            except (error.URLError, ElementTree.ParseError, ValueError, TimeoutError):
                pass

            if self.sec_user_agent:
                try:
                    if cik_by_symbol is None:
                        payload = self._fetch_sec_url(
                            "https://www.sec.gov/files/company_tickers.json"
                        )
                        records = json.loads(payload.decode("utf-8"))
                        if isinstance(records, dict):
                            records = records.values()
                        cik_by_symbol = {
                            str(record["ticker"]).upper(): str(record["cik_str"]).zfill(10)
                            for record in records
                            if isinstance(record, dict) and record.get("ticker") and record.get("cik_str")
                        }
                    cik = cik_by_symbol.get(symbol)
                    if cik:
                        articles.extend(parse_sec_atom(self._fetch_sec(symbol, cik), symbol))
                except (error.URLError, ElementTree.ParseError, ValueError, TimeoutError, KeyError):
                    cik_by_symbol = cik_by_symbol or {}
        return articles


class FeedRegistry:
    def __init__(self, sources: Iterable[NewsSource] | None = None) -> None:
        self._sources: list[NewsSource] = list(sources or ())

    def register(self, source: NewsSource) -> None:
        self._sources.append(source)

    def fetch_articles(self) -> list[dict[str, object]]:
        articles: list[dict[str, object]] = []
        for source in self._sources:
            for item in source.fetch_articles():
                if not isinstance(item, dict):
                    continue
                articles.append(normalize_news_article(item, default_source=getattr(source, "__class__", type(source)).__name__))
        return articles


def build_default_news_sources(symbols: Iterable[str] | None = None) -> FeedRegistry:
    configured_symbols = tuple(symbols or ())
    sources: list[NewsSource] = []

    api_key = os.getenv("FINNHUB_API_KEY")
    if api_key:
        sources.append(FinnhubNewsSource(api_key=api_key, symbols=configured_symbols))

from __future__ import annotations

import io
import json
import logging
import sys
from collections import Counter
from contextlib import closing, redirect_stdout
from datetime import datetime, timedelta, timezone
from pathlib import Path

if __package__ in (None, ""):
    sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from src.db import DatabasePath, _connect, create_tables, save_daily_report

logger = logging.getLogger(__name__)
PROJECT_ROOT = Path(__file__).resolve().parents[1]
REPORTS_DIR = PROJECT_ROOT / "reports"
DEFAULT_DB_PATH = PROJECT_ROOT / "data" / "research.db"
STARTING_CASH = 10_000.0


def _timestamp(value: str) -> datetime:
    if value.endswith("Z"):
        value = value[:-1] + "+00:00"
    parsed = datetime.fromisoformat(value)
    if parsed.tzinfo is None or parsed.utcoffset() is None:
        parsed = parsed.replace(tzinfo=timezone.utc)
    return parsed.astimezone(timezone.utc)


def _apply_fill(cash: float, positions: dict[str, int], order) -> float:
    quantity = int(order["qty"])
    fill_price = float(order["fill_price"])
    if order["side"] == "BUY":
        cash -= quantity * fill_price
        positions[order["symbol"]] = positions.get(order["symbol"], 0) + quantity
    elif order["side"] == "SELL":
        cash += quantity * fill_price
        positions[order["symbol"]] = positions.get(order["symbol"], 0) - quantity
    return cash


def _equity(cash: float, positions: dict[str, int], provider: PriceProvider, when: datetime):
    value = cash
    for symbol, quantity in positions.items():
        if quantity:
            price = provider.get_price(symbol, when)
            if price is None:
                return None
            value += quantity * price
    return value


def _build_report(db_path: DatabasePath, provider: PriceProvider, now: datetime) -> str:
    day = now.date()
    start = datetime.combine(day, datetime.min.time(), timezone.utc)
    end = start + timedelta(days=1)
    with closing(_connect(db_path)) as connection:
        signals = connection.execute(
            "SELECT id, symbol, signal, created_at FROM signals ORDER BY id"
        ).fetchall()
        orders = connection.execute(
            "SELECT symbol, side, qty, fill_price, status, created_at FROM orders ORDER BY id"
        ).fetchall()
        events = connection.execute(
            "SELECT approved, reasons, created_at FROM risk_events ORDER BY id"
        ).fetchall()
        resolved = connection.execute("SELECT COUNT(*) FROM outcomes").fetchone()[0]

    def is_today(value: str) -> bool:
        try:
            return start <= _timestamp(value) < end
        except (TypeError, ValueError):
            return False

    today_signals = [row for row in signals if is_today(row["created_at"])]
    today_orders = [row for row in orders if is_today(row["created_at"])]
    today_events = [row for row in events if is_today(row["created_at"])]
    signal_counts = Counter(row["signal"] for row in today_signals)
    symbol_counts = Counter(row["symbol"] for row in today_signals)
    fills = [row for row in today_orders if row["status"] == "filled"]
    approvals = sum(bool(row["approved"]) for row in today_events)
    rejections = len(today_events) - approvals
    rejection_reasons: Counter[str] = Counter()
    loss_cap_hit = False
    for event in today_events:
        if event["approved"]:
            continue
        try:
            reasons = json.loads(event["reasons"])
        except (TypeError, ValueError):
            reasons = ["unreadable rejection reason"]
        for reason in reasons or ["unspecified rejection"]:
            rejection_reasons[str(reason)] += 1
            if "daily loss cap has been hit" in str(reason).lower():
                loss_cap_hit = True

    cash_at_start = STARTING_CASH
    cash_now = STARTING_CASH
    positions_at_start: dict[str, int] = {}
    positions_now: dict[str, int] = {}
    for order in orders:
        if order["status"] != "filled" or order["fill_price"] is None:
            continue
        order_time = _timestamp(order["created_at"])
        if order_time > now:
            continue
        cash_now = _apply_fill(cash_now, positions_now, order)
        if order_time < start:
            cash_at_start = _apply_fill(cash_at_start, positions_at_start, order)
    equity_start = _equity(cash_at_start, positions_at_start, provider, start)
    equity_now = _equity(cash_now, positions_now, provider, now)

    output = io.StringIO()
    with redirect_stdout(output):
        print_report(db_path)
    full_summary = output.getvalue()
    confidence_summary = full_summary.split("Resolved outcomes by confidence:\n", 1)
    bucket_summary = confidence_summary[-1].strip()

    lines = [
        f"Nightly review for {day.isoformat()}",
        "",
        "Signals today: " + ", ".join(f"{kind}={signal_counts.get(kind, 0)}" for kind in ("BUY", "SELL", "HOLD")),
        f"Orders today: {len(today_orders)}; fills: {len(fills)}",
        "Risk rejections by reason:",
    ]
    lines.extend(f"  {reason}: {count}" for reason, count in sorted(rejection_reasons.items()))
    if not rejection_reasons:
        lines.append("  None")
    if equity_start is None or equity_now is None:
        lines.append("Account equity vs start of day: unavailable (missing price data).")
    else:
        change = equity_now - equity_start
        lines.append(
            f"Account equity vs start of day: ${equity_now:,.2f} vs ${equity_start:,.2f} "
            f"(change ${change:+,.2f})."
        )
    lines.extend((f"Resolved outcomes so far: {resolved}", "", "Confidence-bucket summary:", bucket_summary, "", "Warnings:"))

    warnings = []
    if not today_signals:
        warnings.append("No signals were logged today.")
    if today_signals and symbol_counts:
        symbol, count = symbol_counts.most_common(1)[0]
        if count / len(today_signals) > 0.5:
            warnings.append(f"More than 50% of today's signals are for {symbol}.")
    latest = getattr(provider, "latest_timestamp", None)
    if latest is not None:
        if latest.tzinfo is None or latest.utcoffset() is None:
            raise ValueError("price provider latest_timestamp must be timezone-aware")
        if now.astimezone(timezone.utc) - latest.astimezone(timezone.utc) > timedelta(days=1):
            warnings.append("The latest price data is older than 1 day.")
    if (PROJECT_ROOT / "KILL").exists():
        warnings.append("The KILL switch file exists.")
    if loss_cap_hit:
        warnings.append("The daily loss cap was hit today.")
    if rejections > 0 and rejections >= approvals * 3:
        warnings.append("Risk rejections outnumber approvals by 3 to 1 or more today.")
    lines.extend(warnings or ["None."])
    return "\n".join(lines) + "\n"

import logging
import math
from contextlib import closing
from datetime import datetime, timedelta, timezone

from .db import DatabasePath, _connect, create_tables

logger = logging.getLogger(__name__)
HORIZONS = {"1d": timedelta(days=1), "5d": timedelta(days=5), "20d": timedelta(days=20)}


def _signal_time(value: str) -> datetime:
    if value.endswith("Z"):
        value = value[:-1] + "+00:00"
    timestamp = datetime.fromisoformat(value)
    if timestamp.tzinfo is None or timestamp.utcoffset() is None:
        timestamp = timestamp.replace(tzinfo=timezone.utc)
    return timestamp.astimezone(timezone.utc)


def process_outcomes(
    db_path: DatabasePath,
    price_provider: PriceProvider,
    now: datetime,
) -> int:
    if now.tzinfo is None or now.utcoffset() is None:
        raise ValueError("now must be a timezone-aware datetime")
    now_utc = now.astimezone(timezone.utc)
    create_tables(db_path)
    processed = 0

    with closing(_connect(db_path)) as connection:
        with connection:
            connection.execute("BEGIN IMMEDIATE")
            signals = connection.execute(
                """
                SELECT s.id, s.symbol, s.signal, s.horizon, s.created_at
                FROM signals AS s
                LEFT JOIN outcomes AS o
                    ON o.signal_id = s.id AND o.horizon_checked = s.horizon
                WHERE o.id IS NULL
                ORDER BY s.id
                """
            ).fetchall()

            for signal in signals:
                horizon = HORIZONS.get(signal["horizon"])
                if horizon is None:
                    logger.warning("Skipping signal %s: unsupported horizon", signal["id"])
                    continue
                try:
                    start_time = _signal_time(signal["created_at"])
                except (TypeError, ValueError) as error:
                    logger.warning("Skipping signal %s: invalid signal time: %s", signal["id"], error)
                    continue
                end_time = start_time + horizon
                if end_time > now_utc:
                    continue

                try:
                    start_price = price_provider.get_price(signal["symbol"], start_time)
                    end_price = price_provider.get_price(signal["symbol"], end_time)
                except Exception as error:
                    logger.warning("Skipping signal %s: price lookup failed: %s", signal["id"], error)
                    continue
                if start_price is None or end_price is None:
                    missing = "signal-time" if start_price is None else "horizon"
                    if start_price is None and end_price is None:
                        missing = "signal-time and horizon"
                    logger.warning("Skipping signal %s: missing %s price", signal["id"], missing)
                    continue
                try:
                    start_price = float(start_price)
                    end_price = float(end_price)
                except (TypeError, ValueError):
                    logger.warning("Skipping signal %s: price is not numeric", signal["id"])
                    continue
                if not all(math.isfinite(price) and price > 0 for price in (start_price, end_price)):
                    logger.warning("Skipping signal %s: price is not positive and finite", signal["id"])
                    continue

                signal_type = signal["signal"]
                if signal_type == "HOLD":
                    return_pct, no_trade = 0.0, 1
                elif signal_type == "BUY":
                    return_pct, no_trade = (end_price - start_price) / start_price, 0
                elif signal_type == "SELL":
                    return_pct, no_trade = (start_price - end_price) / start_price, 0
                else:
                    logger.warning("Skipping signal %s: unsupported signal type", signal["id"])
                    continue

                connection.execute(
                    """
                    INSERT INTO outcomes (
                        signal_id, horizon_checked, price_at_check, return_pct,
                        no_trade, checked_at
                    ) VALUES (?, ?, ?, ?, ?, ?)
                    """,
                    (
                        signal["id"],
                        signal["horizon"],
                        end_price,
                        return_pct,
                        no_trade,
                        now_utc.isoformat().replace("+00:00", "Z"),
                    ),
                )
                processed += 1
    return processed

import csv
import math
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from pathlib import Path
from typing import Protocol


@dataclass(frozen=True)
class Quote:
    price: float
    timestamp: datetime


class PriceProvider(Protocol):
    def get_quote(self, symbol: str, when: datetime) -> Quote | None:
        """Return the latest quote no later than when."""
        ...

    def get_price(self, symbol: str, when: datetime) -> float | None:
        """Return the latest known price no later than when."""
        ...


class CsvPriceProvider:
    def __init__(self, csv_path: str | Path = "data/prices.csv") -> None:
        self._prices: dict[str, list[tuple[datetime, float]]] = {}
        with Path(csv_path).open(newline="", encoding="utf-8") as csv_file:
            reader = csv.DictReader(csv_file)
            if not {"timestamp", "symbol", "price"} <= set(reader.fieldnames or ()):
                raise ValueError("CSV must contain timestamp, symbol, and price columns")
            for row in reader:
                timestamp_text = row["timestamp"].strip()
                if timestamp_text.endswith("Z"):
                    timestamp_text = timestamp_text[:-1] + "+00:00"
                timestamp = datetime.fromisoformat(timestamp_text)
                if timestamp.utcoffset() != timedelta(0):
                    raise ValueError("price timestamps must be UTC")
                symbol = row["symbol"].strip()
                price = float(row["price"])
                if not symbol or not math.isfinite(price) or price <= 0:
                    raise ValueError("CSV contains an invalid symbol or price")
                self._prices.setdefault(symbol, []).append(
                    (timestamp.astimezone(timezone.utc), price)
                )

        for prices in self._prices.values():
            prices.sort(key=lambda quote: quote[0])

    @property
    def latest_timestamp(self) -> datetime | None:
        return max((quotes[-1][0] for quotes in self._prices.values() if quotes), default=None)

    def get_quote(self, symbol: str, when: datetime) -> Quote | None:
        if when.tzinfo is None or when.utcoffset() is None:
            raise ValueError("when must be a timezone-aware datetime")
        target = when.astimezone(timezone.utc)
        last_quote = None
        for timestamp, price in self._prices.get(symbol, ()):
            if timestamp > target:
                break
            last_quote = Quote(price, timestamp)
        return last_quote

    def get_price(self, symbol: str, when: datetime) -> float | None:
        quote = self.get_quote(symbol, when)
        return None if quote is None else quote.price

def summarize_confidence_outcomes(rows):
    summaries = []
    for signal_type in ("BUY", "SELL"):
        for label, lower, upper in BUCKETS:
            returns = [
                row[2]
                for row in rows
                if row[0] == signal_type
                and lower <= row[1]
                and (row[1] < upper or (upper == 1.0 and row[1] <= upper))
            ]
            count = len(returns)
            if count < MIN_BUCKET_OUTCOMES:
                summary = f"insufficient data (n={count})"
                average = None
                hit_rate = None
            else:
                average = sum(returns) / count
                hit_rate = sum(value > 0 for value in returns) / count
                summary = f"n={count}, average return={average:.2%}, hit rate={hit_rate:.1%}"
            summaries.append(
                {
                    "signal": signal_type,
                    "confidence_bucket": label,
                    "resolved_count": count,
                    "average_return": average,
                    "hit_rate": hit_rate,
                    "summary": summary,
                }
            )
    return summaries

def _recent_news_context(
    db_path: DatabasePath,
    symbol: str,
    now: datetime,
) -> tuple[float, int]:
    cutoff = now - timedelta(hours=48)
    with sqlite3.connect(db_path) as connection:
        rows = connection.execute(
            "SELECT sentiment, relevance_score, quality_score, published_at "
            "FROM news_events WHERE symbol = ? AND published_at IS NOT NULL",
            (symbol,),
        ).fetchall()

    weighted_sentiment = 0.0
    total_weight = 0.0
    item_count = 0
    for sentiment, relevance, quality, published_at in rows:
        try:
            text = str(published_at).replace("Z", "+00:00")
            timestamp = datetime.fromisoformat(text)
            if timestamp.tzinfo is None or timestamp.utcoffset() is None:
                continue
            timestamp = timestamp.astimezone(timezone.utc)
            if timestamp < cutoff or timestamp > now:
                continue
            score = max(-1.0, min(1.0, (float(sentiment) - 0.5) * 2.0))
            weight = max(0.0, float(relevance)) * max(0.0, float(quality))
        except (TypeError, ValueError):
            continue
        if weight <= 0:
            continue
        weighted_sentiment += score * weight
        total_weight += weight
        item_count += 1
    if not total_weight:
        return 0.0, 0
    return weighted_sentiment / total_weight, item_count


def _collect_history(
    price_provider: PriceProviderProtocol,
    symbol: str,
    now: datetime,
    lookback_days: int,
) -> list[tuple[datetime, float]]:
    history: list[tuple[datetime, float]] = []
    for offset in range(lookback_days + 1):
        when = now - timedelta(days=offset)
        price = price_provider.get_price(symbol, when)
        if price is not None:
            history.append((when, float(price)))
    history.sort(key=lambda pair: pair[0])
    return history

def summarize_strategy_performance(db_path: DatabasePath = "data/research.db") -> list[dict[str, object]]:
    path = Path(db_path)
    if not path.exists():
        return []

    with sqlite3.connect(path) as connection:
        connection.row_factory = sqlite3.Row
        tables = {
            row[0]
            for row in connection.execute(
                "SELECT name FROM sqlite_master WHERE type='table'"
            )
        }
        if "signals" not in tables:
            return []

        rows = connection.execute(
            """
            SELECT
                s.model_name,
                COUNT(*) AS signal_count,
                AVG(s.confidence) AS average_confidence,
                SUM(CASE WHEN s.signal = 'BUY' THEN 1 ELSE 0 END) AS buy_count,
                SUM(CASE WHEN s.signal = 'SELL' THEN 1 ELSE 0 END) AS sell_count,
                SUM(CASE WHEN s.signal = 'HOLD' THEN 1 ELSE 0 END) AS hold_count,
                AVG(CASE WHEN o.return_pct IS NULL THEN NULL ELSE o.return_pct END) AS average_return,
                AVG(CASE WHEN o.return_pct IS NOT NULL AND o.return_pct > 0 THEN 1.0 ELSE 0.0 END) AS win_rate
            FROM signals AS s
            LEFT JOIN outcomes AS o ON o.signal_id = s.id
            GROUP BY s.model_name
            ORDER BY signal_count DESC, average_confidence DESC, average_return DESC NULLS LAST
            """
        ).fetchall()

    summary: list[dict[str, object]] = []
    for row in rows:
        summary.append(
            {
                "model_name": row["model_name"],
                "signal_count": int(row["signal_count"] or 0),
                "buy_count": int(row["buy_count"] or 0),
                "sell_count": int(row["sell_count"] or 0),
                "hold_count": int(row["hold_count"] or 0),
                "average_confidence": float(row["average_confidence"] or 0.0),
                "average_return": None if row["average_return"] is None else float(row["average_return"]),
                "win_rate": None if row["win_rate"] is None else float(row["win_rate"]),
            }
        )
    return summary


def summarize_strategy_drift(db_path: DatabasePath = "data/research.db") -> list[dict[str, object]]:
    path = Path(db_path)
    if not path.exists():
        return []

    with sqlite3.connect(path) as connection:
        connection.row_factory = sqlite3.Row
        tables = {
            row[0]
            for row in connection.execute(
                "SELECT name FROM sqlite_master WHERE type='table'"
            )
        }
        if "signals" not in tables:
            return []

        rows = connection.execute(
            """
            SELECT model_name, confidence, created_at
            FROM signals
            ORDER BY created_at DESC, id DESC
            """
        ).fetchall()

    by_model: dict[str, list[tuple[float, datetime]]] = {}
    for row in rows:
        model_name = row["model_name"]
        created_at = row["created_at"]
        if created_at is None:
            continue
        try:
            created_dt = datetime.fromisoformat(created_at.replace("Z", "+00:00"))
        except ValueError:
            continue
        by_model.setdefault(model_name, []).append((float(row["confidence"]), created_dt))

    drift_rows: list[dict[str, object]] = []
    for model_name, points in by_model.items():
        if len(points) < 2:
            continue
        recent_points = points[: max(1, len(points) // 2)]
        older_points = points[max(1, len(points) // 2):]
        recent_avg = sum(value for value, _ in recent_points) / len(recent_points)
        older_avg = sum(value for value, _ in older_points) / len(older_points) if older_points else recent_avg
        confidence_delta = recent_avg - older_avg
        drift_rows.append(
            {
                "model_name": model_name,
                "recent_signal_count": len(recent_points),
                "confidence_delta": confidence_delta,
                "recent_average_confidence": recent_avg,
                "baseline_average_confidence": older_avg,
                "status": "declining" if confidence_delta < 0 else "stable",
            }
        )

    drift_rows.sort(key=lambda row: (row["confidence_delta"], row["recent_signal_count"]))
    return drift_rows


def run_research_cycle(
    *,
    db_path: DatabasePath = "data/research.db",
    price_provider: PriceProviderProtocol,
    config: ResearchConfig | None = None,
    now: datetime | None = None,
    registry: StrategyRegistry | None = None,
    return_summary: bool = False,
) -> list[dict[str, object]] | dict[str, object]:
    resolved_config = config or ResearchConfig()
    resolved_now = now or datetime.now(timezone.utc)
    resolved_registry = registry or _fallback_registry()

    db_path = Path(db_path)
    db_path.parent.mkdir(parents=True, exist_ok=True)
    create_tables(db_path)

    inserted: list[dict[str, object]] = []
    per_symbol_count: dict[str, int] = {}

    for symbol in resolved_config.symbols:
        history = _collect_history(
            price_provider,
            symbol,
            resolved_now,
            resolved_config.lookback_days,
        )
        if not history:
            continue

        for strategy_name in resolved_config.strategy_names:
            if strategy_name not in resolved_registry.names:
                continue

            strategy = resolved_registry.get(strategy_name)
            signal = strategy.generator(symbol, history, resolved_now)
            if signal is None:
                continue
            news_sentiment, news_count = _recent_news_context(db_path, symbol, resolved_now)
            direction = 1.0 if signal.signal == "BUY" else -1.0 if signal.signal == "SELL" else 0.0
            confidence_adjustment = max(-0.1, min(0.1, news_sentiment * direction * 0.1))
            adjusted_confidence = max(0.0, min(1.0, signal.confidence + confidence_adjustment))
            if adjusted_confidence < resolved_config.min_confidence:
                continue
            if per_symbol_count.get(symbol, 0) >= resolved_config.max_signals_per_symbol:
                continue

            payload = signal.as_payload()
            payload["confidence"] = adjusted_confidence
            if news_count:
                tone = "positive" if news_sentiment > 0.05 else "negative" if news_sentiment < -0.05 else "neutral"
                alignment = "aligned with" if news_sentiment * direction > 0.05 else "conflicted with" if news_sentiment * direction < -0.05 else "did not materially affect"
                payload["rationale"] = (
                    f"{signal.rationale} Recent news context: {tone} sentiment {alignment} "
                    f"the {signal.signal} signal across {news_count} item(s); confidence adjustment "
                    f"{confidence_adjustment:+.3f}."
                )
            payload["horizon"] = signal.horizon
            payload["created_at"] = resolved_now.isoformat()
            with sqlite3.connect(db_path) as connection:
                duplicate = connection.execute(
                    "SELECT 1 FROM signals WHERE symbol = ? AND signal = ? "
                    "AND confidence = ? AND horizon = ? AND rationale = ? "
                    "AND model_name = ? AND prompt_version = ? "
                    "AND price_at_signal IS ? AND date(created_at) = ? LIMIT 1",
                    (
                        payload["symbol"],
                        payload["signal"],
                        payload["confidence"],
                        payload["horizon"],
                        payload["rationale"],
                        payload["model_name"],
                        payload["prompt_version"],
                        payload["price_at_signal"],
                        resolved_now.date().isoformat(),
                    ),
                ).fetchone()
            if duplicate is not None:
                continue
            signal_id = insert_signal(
                db_path,
                item_id=None,
                symbol=str(payload["symbol"]),
                signal=str(payload["signal"]),
                confidence=float(payload["confidence"]),
                horizon=str(payload["horizon"]),
                rationale=str(payload["rationale"]),
                model_name=str(payload["model_name"]),
                prompt_version=str(payload["prompt_version"]),
                price_at_signal=payload["price_at_signal"],
                created_at=str(payload["created_at"]),
            )
            payload["id"] = signal_id
            inserted.append(payload)
            per_symbol_count[symbol] = per_symbol_count.get(symbol, 0) + 1

    result: dict[str, object] = {
        "signals_generated": len(inserted),
        "signals": inserted,
        "strategy_ranking": summarize_strategy_performance(db_path),
        "strategy_drift": summarize_strategy_drift(db_path),
    }
    if return_summary:
        return result
    return inserted

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, time
from decimal import Decimal, InvalidOperation, ROUND_CEILING, ROUND_FLOOR
from zoneinfo import ZoneInfo


@dataclass(frozen=True)
class RiskConfig:
    max_position_pct: float = 0.05
    max_total_exposure_pct: float = 0.50
    daily_loss_cap_pct: float = 0.02
    max_data_age_seconds: int = 60
    min_confidence: float = 0.6
    symbol_whitelist: frozenset[str] = frozenset()
    allow_short: bool = False

    def __post_init__(self) -> None:
        if isinstance(self.symbol_whitelist, str):
            raise ValueError("symbol_whitelist must be an iterable of non-empty symbol strings")
        try:
            symbols = tuple(self.symbol_whitelist)
        except TypeError as exc:
            raise ValueError("symbol_whitelist must be an iterable of non-empty symbol strings") from exc
        if any(not isinstance(symbol, str) or not symbol.strip() for symbol in symbols):
            raise ValueError("symbol_whitelist must contain non-empty symbol strings")
        object.__setattr__(self, "symbol_whitelist", frozenset(symbol.strip() for symbol in symbols))


@dataclass
class ProposedTrade:
    symbol: str
    side: str
    price: float
    confidence: float
    price_timestamp: datetime


@dataclass
class AccountState:
    equity: float
    day_start_equity: float
    cash: float
    positions: dict[str, float]
    as_of: datetime


@dataclass
class RiskDecision:
    approved: bool
    qty: int
    reasons: list[str]


def _decimal(value: object, name: str) -> Decimal:
    if isinstance(value, bool) or not isinstance(value, (int, float, Decimal)):
        raise ValueError(f"{name} must be a number")
    try:
        result = Decimal(str(value))
    except (InvalidOperation, ValueError):
        raise ValueError(f"{name} must be a finite number") from None
    if not result.is_finite():
        raise ValueError(f"{name} must be a finite number")
    return result


def _aware_datetime(value: object, name: str) -> datetime:
    if not isinstance(value, datetime) or value.tzinfo is None or value.utcoffset() is None:
        raise ValueError(f"{name} must be a timezone-aware datetime")
    return value


def _floor_ratio(numerator: Decimal, denominator: Decimal) -> int:
    return int((numerator / denominator).to_integral_value(rounding=ROUND_FLOOR))


def _ceil_ratio(numerator: Decimal, denominator: Decimal) -> int:
    return int((numerator / denominator).to_integral_value(rounding=ROUND_CEILING))


def _validate_config(config: RiskConfig) -> tuple[Decimal, Decimal, Decimal, Decimal, Decimal]:
    if not isinstance(config, RiskConfig):
        raise ValueError("config must be a RiskConfig")

    max_position = _decimal(config.max_position_pct, "max_position_pct")
    max_exposure = _decimal(config.max_total_exposure_pct, "max_total_exposure_pct")
    loss_cap = _decimal(config.daily_loss_cap_pct, "daily_loss_cap_pct")
    max_age = _decimal(config.max_data_age_seconds, "max_data_age_seconds")
    min_confidence = _decimal(config.min_confidence, "min_confidence")

    if not Decimal(0) < max_position <= Decimal(1):
        raise ValueError("max_position_pct must be greater than 0 and at most 1")
    if not Decimal(0) < max_exposure <= Decimal(1):
        raise ValueError("max_total_exposure_pct must be greater than 0 and at most 1")
    if not Decimal(0) < loss_cap <= Decimal(1):
        raise ValueError("daily_loss_cap_pct must be greater than 0 and at most 1")
    if max_age <= 0:
        raise ValueError("max_data_age_seconds must be greater than 0")
    if not Decimal(0) < min_confidence <= Decimal(1):
        raise ValueError("min_confidence must be greater than 0 and at most 1")
    if not isinstance(config.symbol_whitelist, frozenset) or any(
        not isinstance(symbol, str) or not symbol.strip()
        for symbol in config.symbol_whitelist
    ):
        raise ValueError("symbol_whitelist must contain non-empty symbol strings")
    if not isinstance(config.allow_short, bool):
        raise ValueError("allow_short must be a boolean")

    return max_position, max_exposure, loss_cap, max_age, min_confidence


def _check_trade(
    trade: ProposedTrade,
    account: AccountState,
    config: RiskConfig,
    now: datetime,
) -> RiskDecision:
    if not isinstance(trade, ProposedTrade):
        raise ValueError("trade must be a ProposedTrade")
    if not isinstance(account, AccountState):
        raise ValueError("account must be an AccountState")

    max_position_pct, max_exposure_pct, loss_cap_pct, max_age, min_confidence = (
        _validate_config(config)
    )
    now = _aware_datetime(now, "now")

    if not isinstance(trade.symbol, str) or not trade.symbol.strip():
        raise ValueError("symbol must be a non-empty string")
    if trade.side not in ("BUY", "SELL"):
        raise ValueError("side must be BUY or SELL")
    price = _decimal(trade.price, "price")
    confidence = _decimal(trade.confidence, "confidence")
    if price <= 0:
        raise ValueError("price must be greater than 0")
    if not Decimal(0) < confidence <= Decimal(1):
        raise ValueError("confidence must be greater than 0 and at most 1")

    price_timestamp = _aware_datetime(trade.price_timestamp, "price_timestamp")
    equity = _decimal(account.equity, "equity")
    day_start_equity = _decimal(account.day_start_equity, "day_start_equity")
    cash = _decimal(account.cash, "cash")
    if equity <= 0:
        raise ValueError("equity must be greater than 0")
    if day_start_equity <= 0:
        raise ValueError("day_start_equity must be greater than 0")
    if cash <= 0:
        raise ValueError("cash must be greater than 0")
    if not isinstance(account.positions, dict):
        raise ValueError("positions must be a dictionary")

    positions: dict[str, Decimal] = {}
    for symbol, market_value in account.positions.items():
        if not isinstance(symbol, str) or not symbol.strip():
            raise ValueError("position symbols must be non-empty strings")
        value = _decimal(market_value, f"position value for {symbol}")
        if value == 0:
            raise ValueError(f"position value for {symbol} must not be zero")
        if value < 0 and not config.allow_short:
            raise ValueError(f"short position in {symbol} is not allowed")
        positions[symbol] = value

    account_timestamp = _aware_datetime(account.as_of, "account.as_of")
    for timestamp, name in (
        (price_timestamp, "price data"),
        (account_timestamp, "account data"),
    ):
        age_seconds = Decimal(str((now - timestamp).total_seconds()))
        if age_seconds < 0:
            raise ValueError(f"{name} timestamp is in the future")
        if age_seconds > max_age:
            raise ValueError(f"{name} is stale")

    if trade.symbol not in config.symbol_whitelist:
        return RiskDecision(False, 0, ["symbol is not on the whitelist"])
    if confidence < min_confidence:
        return RiskDecision(False, 0, ["confidence is below the minimum"])

    market_now = now.astimezone(ZoneInfo("America/New_York"))
    # US market holidays are not handled yet.
    if market_now.weekday() >= 5 or not (
        time(9, 30) <= market_now.time().replace(tzinfo=None) < time(16, 0)
    ):
        return RiskDecision(False, 0, ["outside regular US market hours"])

    daily_loss_pct = (day_start_equity - equity) / day_start_equity
    if daily_loss_pct >= loss_cap_pct:
        return RiskDecision(False, 0, ["daily loss cap has been hit"])

    current_position = positions.get(trade.symbol, Decimal(0))
    if trade.side == "SELL" and current_position <= 0 and not config.allow_short:
        return RiskDecision(False, 0, ["cannot sell a position that is not held"])

    position_limit = equity * max_position_pct
    total_limit = equity * max_exposure_pct
    other_exposure = sum(
        (abs(value) for symbol, value in positions.items() if symbol != trade.symbol),
        Decimal(0),
    )
    remaining_total = total_limit - other_exposure
    if remaining_total < 0:
        return RiskDecision(False, 0, ["existing positions exceed the total exposure limit"])
    allowed_position_value = min(position_limit, remaining_total)

    if trade.side == "BUY":
        minimum_qty = _ceil_ratio(-allowed_position_value - current_position, price)
        maximum_qty = _floor_ratio(allowed_position_value - current_position, price)
        maximum_qty = min(maximum_qty, _floor_ratio(cash, price))
    else:
        minimum_qty = _ceil_ratio(current_position - allowed_position_value, price)
        maximum_qty = _floor_ratio(current_position + allowed_position_value, price)
        if not config.allow_short:
            maximum_qty = min(maximum_qty, _floor_ratio(current_position, price))

    minimum_qty = max(1, minimum_qty)
    if maximum_qty < minimum_qty:
        if trade.side == "BUY" and cash < price:
            reason = "insufficient cash for one share"
        else:
            reason = "no whole-share quantity fits within the risk limits"
        return RiskDecision(False, 0, [reason])

    return RiskDecision(True, maximum_qty, [])


def check_trade(
    trade: ProposedTrade,
    account: AccountState,
    config: RiskConfig,
    now: datetime,
) -> RiskDecision:
    try:
        return _check_trade(trade, account, config, now)
    except ValueError as error:
        return RiskDecision(False, 0, [str(error)])
    except Exception as error:
        return RiskDecision(
            False,
            0,
            [f"risk check failed closed: invalid or ambiguous input ({type(error).__name__})"],
        )

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path
from typing import Callable, Protocol

from .research_runner import ResearchConfig, run_research_cycle
from .strategies import StrategyRegistry, get_default_registry


class PriceProviderLike(Protocol):
    def get_price(self, symbol: str, when: datetime) -> float | None:
        ...


@dataclass
class ResearchScheduler:
    db_path: str | Path = "data/research.db"
    price_provider: PriceProviderLike | None = None
    config: ResearchConfig = field(default_factory=ResearchConfig)
    registry: StrategyRegistry | None = None
    now_factory: Callable[[], datetime] = lambda: datetime.now(timezone.utc)
    interval_seconds: float = 3600.0
    last_cycle_at: datetime | None = None

    def __post_init__(self) -> None:
        if self.price_provider is None:
            raise ValueError("price_provider is required")
        if self.registry is None:
            self.registry = get_default_registry()
        if self.interval_seconds <= 0:
            raise ValueError("interval_seconds must be positive")

    def run_once(self, *, now: datetime | None = None) -> dict[str, object]:
        resolved_now = now or self.now_factory()
        db_path = Path(self.db_path)
        db_path.parent.mkdir(parents=True, exist_ok=True)
        result = run_research_cycle(
            db_path=db_path,
            price_provider=self.price_provider,
            config=self.config,
            now=resolved_now,
            registry=self.registry,
            return_summary=True,
        )
        self.last_cycle_at = resolved_now
        return result

    def run_if_due(self, *, now: datetime | None = None) -> dict[str, object] | None:
        resolved_now = now or self.now_factory()
        if self.last_cycle_at is not None:
            elapsed = (resolved_now - self.last_cycle_at).total_seconds()
            if elapsed < self.interval_seconds:
                return None
        return self.run_once(now=resolved_now)

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from typing import Callable, Iterable


@dataclass(frozen=True)
class ResearchSignal:
    symbol: str
    signal: str
    confidence: float
    horizon: str
    rationale: str
    model_name: str
    prompt_version: str = "research-v1"
    price_at_signal: float | None = None

    def as_payload(self) -> dict[str, object]:
        return {
            "symbol": self.symbol,
            "signal": self.signal,
            "confidence": float(self.confidence),
            "horizon": self.horizon,
            "rationale": self.rationale,
            "model_name": self.model_name,
            "prompt_version": self.prompt_version,
            "price_at_signal": self.price_at_signal,
        }


@dataclass(frozen=True)
class StrategySpec:
    name: str
    description: str
    generator: Callable[[str, list[tuple[datetime, float]], datetime], ResearchSignal | None]


class StrategyRegistry:
    def __init__(self, strategies: Iterable[StrategySpec] | None = None) -> None:
        self._strategies: dict[str, StrategySpec] = {}
        for strategy in strategies or ():
            self.register(strategy)

    def register(self, strategy: StrategySpec) -> None:
        self._strategies[strategy.name] = strategy

    @property
    def names(self) -> tuple[str, ...]:
        return tuple(self._strategies.keys())

    def get(self, name: str) -> StrategySpec:
        return self._strategies[name]

    def iterate(self) -> tuple[StrategySpec, ...]:
        return tuple(self._strategies.values())


def _clamp(value: float, low: float, high: float) -> float:
    return max(low, min(value, high))


def _price_history_summary(history: list[tuple[datetime, float]], now: datetime) -> tuple[float, float, float]:
    if not history:
        raise ValueError("price history is empty")
    current_price = history[-1][1]
    if len(history) >= 5:
        short_price = history[-5][1]
        short_return = (current_price - short_price) / short_price if short_price else 0.0
    else:
        short_return = 0.0
    if len(history) >= 20:
        long_price = history[-20][1]
        long_return = (current_price - long_price) / long_price if long_price else 0.0
    else:
        long_return = short_return
    average_price = sum(price for _, price in history) / len(history)
    return current_price, short_return, long_return, average_price


def momentum_strategy(symbol: str, history: list[tuple[datetime, float]], now: datetime) -> ResearchSignal | None:
    if len(history) < 2:
        return None
    current_price = history[-1][1]
    short_window = history[-5][1] if len(history) >= 5 else history[0][1]
    long_window = history[-20][1] if len(history) >= 20 else history[0][1]
    short_return = (current_price - short_window) / short_window if short_window else 0.0
    long_return = (current_price - long_window) / long_window if long_window else 0.0
    score = float((short_return * 0.6) + (long_return * 0.4))
    if score > 0.04:
        signal = "BUY"
        direction = "uptrend momentum remains positive"
    elif score < -0.04:
        signal = "SELL"
        direction = "downtrend momentum remains negative"
    else:
        signal = "HOLD"
        direction = "price trend is too weak to act on"
    confidence = _clamp(abs(score) * 18.0 + 0.6, 0.6, 0.95)
    return ResearchSignal(
        symbol=symbol,
        signal=signal,
        confidence=confidence,
        horizon="5d",
        rationale=f"Momentum strategy: {direction}; current price {current_price:.2f} vs prior window behavior.",
        model_name="momentum",
        price_at_signal=current_price,
    )


def mean_reversion_strategy(symbol: str, history: list[tuple[datetime, float]], now: datetime) -> ResearchSignal | None:
    if len(history) < 3:
        return None
    current_price = history[-1][1]
    average_price = sum(price for _, price in history) / len(history)
    normalized_gap = (current_price - average_price) / average_price if average_price else 0.0
    if normalized_gap < -0.03:
        signal = "BUY"
        direction = "price is trading below its rolling average"
    elif normalized_gap > 0.03:
        signal = "SELL"
        direction = "price is trading above its rolling average"
    else:
        signal = "HOLD"
        direction = "price remains near its rolling average"
    confidence = _clamp(abs(normalized_gap) * 20.0 + 0.6, 0.6, 0.94)
    return ResearchSignal(
        symbol=symbol,
        signal=signal,
        confidence=confidence,
        horizon="20d",
        rationale=f"Mean reversion strategy: {direction}; average price {average_price:.2f} vs current {current_price:.2f}.",
        model_name="mean_reversion",
        price_at_signal=current_price,
    )


def get_default_registry() -> StrategyRegistry:
    return StrategyRegistry(
        (
            StrategySpec(
                name="momentum",
                description="Trend-following momentum strategy over recent price changes.",
                generator=momentum_strategy,
            ),
            StrategySpec(
                name="mean_reversion",
                description="Mean-reversion strategy based on price vs rolling average.",
                generator=mean_reversion_strategy,
            ),
        )
    )

        for node in ast.walk(tree):

class StaticPrices:
    def get_price(self, symbol, when):
        return 100.0 if symbol == "ACME" else None

    def get_quote(self, symbol, when):
        from src.prices import Quote

        if symbol != "ACME":
            return None
        return Quote(100.0, NOW)


class RecordingBroker(SimBroker):
    def __init__(self):
        super().__init__(StaticPrices(), clock=lambda: NOW)
        self.submissions = []

    def submit_order(self, symbol, side, qty, client_order_id):
        self.submissions.append((symbol, side, qty, client_order_id))
        return super().submit_order(symbol, side, qty, client_order_id)


class FailingBroker(RecordingBroker):
    def submit_order(self, symbol, side, qty, client_order_id):
        self.submissions.append((symbol, side, qty, client_order_id))
        raise RuntimeError("simulated broker failure")

def test_nightly_imports_no_network_ai_or_broker_sdks():
    tree = ast.parse(Path(nightly.__file__).read_text(encoding="utf-8"))
    forbidden = {
        "anthropic", "openai", "requests", "httpx", "alpaca", "alpaca_trade_api",
        "ib_insync", "ibapi", "ccxt", "robin_stocks", "tda_api", "schwab_py",
    }
    imported = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            imported.update(alias.name.split(".")[0] for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            imported.add(node.module.split(".")[0])
    assert imported.isdisjoint(forbidden)

class FakePriceProvider:
    def __init__(self) -> None:
        self._prices = {
            "ACME": [
                (datetime(2026, 1, 1, tzinfo=timezone.utc), 90.0),
                (datetime(2026, 1, 5, tzinfo=timezone.utc), 95.0),
                (datetime(2026, 1, 10, tzinfo=timezone.utc), 102.0),
                (datetime(2026, 1, 15, tzinfo=timezone.utc), 110.0),
            ],
            "BETA": [
                (datetime(2026, 1, 1, tzinfo=timezone.utc), 200.0),
                (datetime(2026, 1, 5, tzinfo=timezone.utc), 205.0),
                (datetime(2026, 1, 10, tzinfo=timezone.utc), 190.0),
                (datetime(2026, 1, 15, tzinfo=timezone.utc), 185.0),
            ],
        }

    def get_price(self, symbol: str, when: datetime):
        prices = self._prices.get(symbol, [])
        last = None
        for timestamp, price in prices:
            if timestamp <= when:
                last = price
            else:
                break
        return last
